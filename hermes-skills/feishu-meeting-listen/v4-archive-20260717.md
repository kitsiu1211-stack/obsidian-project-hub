会议实时语音对话 V4 — BlackHole 音频 → Paraformer ASR → DeepSeek → 豆包 TTS

import os, sys, json, time, uuid, queue, threading, subprocess, asyncio, hashlib

# ── 加载 .env ──
ENV_FILE = os.path.expanduser("~/.hermes/skills/feishu/feishu-meeting-listen/config/.env")

DASHSCOPE_KEY = os.environ.get("DASHSCOPE_API_KEY", "")

DOUBAO_KEY = os.environ.get("DOUBAO_TTS_KEY", "")

CACHE_DIR = os.path.expanduser("~/.hermes/cache/meeting_voice")
os.makedirs(CACHE_DIR, exist_ok=True)

# ── 日志：所有调试输出写文件，stdout 只输出 JSON 状态行 ──
import logging
LOG_FILE = os.path.join(CACHE_DIR, "debug.log")
logging.basicConfig(
    level=logging.DEBUG,

def status(ok, msg="", **kw):
    """stdout 只输出这一行 JSON，不污染用户终端"""
    out = {"ok": ok, "ts": time.time()}
    if msg:
        out["msg"] = msg
    out.update(kw)
    print(json.dumps(out, ensure_ascii=False), flush=True)

SAMPLE_RATE = 16000
CHUNK_SIZE = 3200
CHANNELS = 1
ASR_WS_URL = "wss://dashscope.aliyuncs.com/api-ws/v1/inference"

conversation = []
last_response_time = 0
RESPONSE_COOLDOWN = 2.0

import pyaudio
import websocket

def find_input_device(p):
    for i in range(p.get_device_count()):
        info = p.get_device_info_by_index(i)
        if info["maxInputChannels"] > 0 and "MacBook" in info["name"] and "麦克风" in info["name"]:
            return i, info["name"]
    return None, None

def audio_capture(p, device_index, audio_queue, stop_event):
    stream = p.open(
        format=pyaudio.paInt16, channels=CHANNELS, rate=SAMPLE_RATE,
        input=True, input_device_index=device_index, frames_per_buffer=CHUNK_SIZE,
    )
    log.info("采集启动 (device=%s)", device_index)
    try:
        while not stop_event.is_set():
            data = stream.read(CHUNK_SIZE, exception_on_overflow=False)
            audio_queue.put(data)
    except Exception as e:
        log.error("音频异常: %s", e)
    finally:
        stream.stop_stream()
        stream.close()

class ASRClient:
    def __init__(self, api_key, transcript_queue):
        self.api_key = api_key
        self.transcript_queue = transcript_queue
        self.ws = None
        self.task_id = str(uuid.uuid4())[:8]
        self.started = threading.Event()

    def _on_open(self, ws):
        ws.send(json.dumps({
            "header": {"action": "run-task", "task_id": self.task_id, "streaming": "duplex"},
            "payload": {
                "task_group": "audio", "task": "asr", "function": "recognition",
                "model": "paraformer-realtime-v2",
                "parameters": {"format": "pcm", "sample_rate": SAMPLE_RATE,
                               "enable_intermediate_result": True, "enable_punctuation": True},
                "input": {}
            }
        }))

    def _on_message(self, ws, message):
        try:
            data = json.loads(message)
            h = data.get("header", {})
            if h.get("event") == "task-started":
                self.started.set()
                log.info("✅ ASR connected (task=%s)", self.task_id)
            elif h.get("event") == "result-generated":
                s = data.get("payload", {}).get("output", {}).get("sentence", {})
                text = s.get("text", "")
                if text:
                    self.transcript_queue.put({"text": text, "is_end": s.get("sentence_end", False), "ts": time.time()})
                    suffix = "。" if s.get("sentence_end") else "..."
                    log.debug("🎤 %s%s", text, suffix)
        except: pass

    def _on_error(self, ws, error): pass
    def _on_close(self, ws, code, msg): pass

    def send_audio(self, data):
        if self.ws and self.ws.sock and self.ws.sock.connected:
            try: self.ws.send(data, opcode=websocket.ABNF.OPCODE_BINARY)
            except: pass

    def connect(self):
        self.ws = websocket.WebSocketApp(
            ASR_WS_URL,
            header={"Authorization": f"Bearer {self.api_key}"},
            on_open=self._on_open, on_message=self._on_message,
            on_error=self._on_error, on_close=self._on_close,
        )
        threading.Thread(target=self.ws.run_forever, daemon=True).start()
        return self.started.wait(timeout=15)

    def close(self):
        try: self.ws.send(json.dumps({"header": {"action": "finish-task", "task_id": self.task_id, "streaming": "duplex"}, "payload": {"input": {}}}))
        except: pass
        try: self.ws.close()
        except: pass

def call_deepseek(messages):
    import requests
    try:
        msgs = [{"role": "system", "content": "你是浪子，袁鑫杰的AI销售助手。在飞书会议中旁听并自然对话。当你被叫到名字时，你可以说：自然的、简短的、有趣的话。不要重复自己。最多两句话。用中文口语。"}]
        for role, text in messages[-8:]:
            msgs.append({"role": "user" if role == "user" else "assistant", "content": text})
        resp = requests.post(
            "https://api.deepseek.com/v1/chat/completions",
            headers={"Authorization": f"Bearer {DEEPSEEK_KEY}", "Content-Type": "application/json"},
            json={"model": "deepseek-chat", "messages": msgs, "max_tokens": 150, "temperature": 0.8},
            timeout=15
        )
        return resp.json()["choices"][0]["message"]["content"]
    except Exception as e:
        log.error("LLM err: %s", e)
        return None

async def doubao_tts(text):
    import websockets, copy as cp
    sys.path.insert(0, os.path.expanduser("~/.hermes/scripts"))
    from doubao_tts_proto import (
        MsgType, EventType,
        start_connection, start_session, finish_session, task_request,
        receive_message, wait_for_event,
    )
    headers = {"X-Api-Key": DOUBAO_KEY, "X-Api-Resource-Id": DOUBAO_RESOURCE_ID}
    try:
        log.info("TTS: connecting...")
        async with websockets.connect(
            os.environ.get("DOUBAO_TTS_WS_URL", "wss://openspeech.bytedance.com/api/v3/tts/bidirection"),
            additional_headers=headers, max_size=10*1024*1024
        ) as ws:
            await start_connection(ws)
            await wait_for_event(ws, MsgType.FullServerResponse, EventType.ConnectionStarted)
            log.info("TTS: session start...")
            sid = str(uuid.uuid4())
            base = {"req_params": {"speaker": DOUBAO_SPEAKER, "audio_params": {"format": "mp3", "sample_rate": 24000}}}
            sr = cp.deepcopy(base); sr["event"] = EventType.StartSession
            await start_session(ws, json.dumps(sr).encode(), sid)
            await wait_for_event(ws, MsgType.FullServerResponse, EventType.SessionStarted)
            log.info("TTS: synthesizing '%s'...", text[:20])
            tr = cp.deepcopy(base); tr["event"] = EventType.TaskRequest; tr["req_params"]["text"] = text
            await task_request(ws, json.dumps(tr).encode(), sid)
            await finish_session(ws, sid)
            audio = b""
            while True:
                msg = await receive_message(ws)
                if msg.type == MsgType.AudioOnlyServer: audio += msg.payload
                elif msg.type == MsgType.FullServerResponse and msg.event in (EventType.SessionFinished, EventType.SessionFailed): break
            if audio:
                path = os.path.join(CACHE_DIR, "last_response.wav")
                with open(path, "wb") as f: f.write(audio)
                log.info("TTS ✅ %d bytes", len(audio))
                return path
            else:
                log.warning("TTS ❌ No audio data")
    except Exception as e:
        log.exception("TTS 异常: %s", e)
    return None

def fish_tts(text):
    """Fish Audio TTS — HTTP API"""
    import requests
    key = os.environ.get("FISH_AUDIO_KEY", "")
    if not key:
        return None
    try:
        resp = requests.post(
            "https://api.fish.audio/v1/tts",
            headers={"Authorization": f"Bearer {key}", "Content-Type": "application/json"},
            json={"text": text, "format": "mp3"},
            timeout=30,
        )
        if resp.status_code == 200 and len(resp.content) > 100:
            path = os.path.join(CACHE_DIR, "last_response.wav")
            with open(path, "wb") as f:
                f.write(resp.content)
            log.info("TTS ✅ Fish Audio: %d bytes", len(resp.content))
            return path
        else:
            log.warning("TTS Fish HTTP %s", resp.status_code)
    except Exception as e:
        log.error("TTS Fish err: %s", e)
    return None

def speak(text):
    """Fish Audio TTS → afplay → 会议旁听"""
    global last_response_time
    log.info("SPEAK: %s", text)
    last_response_time = time.time()
    try:
        path = fish_tts(text)
        if path and os.path.getsize(path) > 100:
            subprocess.run(["SwitchAudioSource", "-t", "output", "-s", "会议旁听"], timeout=3, capture_output=True)
            subprocess.run(["afplay", path], timeout=30)
            return True
    except Exception as e:
        log.error("SPEAK err: %s", e)
    return False

def main():
    global conversation, last_response_time

    # 先查活跃会议

    log.info("使用 %s (device=%s)", dev_name, dev_idx)

    audio_queue = queue.Queue()
    transcript_queue = queue.Queue()
    stop_event = threading.Event()

    threading.Thread(target=audio_capture, args=(p, dev_idx, audio_queue, stop_event), daemon=True).start()

    asr = ASRClient(DASHSCOPE_KEY, transcript_queue)
    if not asr.connect():

    log.info("提示: 说「浪子」触发回复。Ctrl+C 退出。")

    def feed_asr():
        while not stop_event.is_set():
            try:
                data = audio_queue.get(timeout=0.5)
                asr.send_audio(data)
            except queue.Empty: continue
            except: break
    threading.Thread(target=feed_asr, daemon=True).start()

    # 会议结束检测线程
    if active_meeting:
        def check_meeting_end():
            last_check = time.time()
            while not stop_event.is_set():
                time.sleep(15)
                try:
                    r = subprocess.run(["/Users/bytedance/.npm-global/bin/lark-cli", "vc", "+meeting-list-active", "--as", "user"],
                                       capture_output=True, text=True, timeout=10)
                    data = json.loads(r.stdout)
                    active_ids = {m["meeting_id"] for m in data.get("data", {}).get("meetings", [])}
                    if active_meeting not in active_ids:
                        log.info("🏁 会议已结束")
                        stop_event.set()
                        return
                except: pass
        threading.Thread(target=check_meeting_end, daemon=True).start()

    try:
        while True:
            try: item = transcript_queue.get(timeout=0.3)
            except queue.Empty: continue

            text = item["text"]
            is_end = item["is_end"]

            if is_end:
                conversation.append(("user", text))
                log.info("[DONE] %s", text)

            keywords = ["浪子", "lanza", "lan zi", "langz", "lang zi", "lancer", "浪", "lang", "mark", "agent", "机器人", "助手"]
            if not any(kw in text.lower() for kw in keywords):
                continue

            if time.time() - last_response_time < RESPONSE_COOLDOWN:
                continue

            resp = call_deepseek(conversation)
            if resp:
                conversation.append(("assistant", resp))
                speak(resp)
    except KeyboardInterrupt:
        log.info("⏹ 停止中...")
    finally:
        stop_event.set(); asr.close(); p.terminate()
        log.info("已退出")

if __name__ == "__main__":
    main()

## 关键修复记录

| Bug | 症状 | 修复 |
|-----|------|------|
| **内容 hash 去重跨会话** | 相同关键词被当成「已回复」跳过，用户说话无响应 | 改为冷却+清缓存 |
| **sentence_end 依赖** | Paraformer 短句不标 sentence_end → 触发逻辑完全跳过 | 不依赖 sentence_end，任意中间结果含关键词即触发 |
| **asyncio.run() 阻塞** | WebSocket 卡在主线程事件循环 | 子线程 asyncio.new_event_loop() |
| **豆包 TTS HTTP 403** | Key 过期 | 切 Fish Audio HTTP API |
| **stdout 污染用户聊天** | print() 调试内容被 Hermes 捕获发送给用户 | 改用 logging → FileHandler |
| **BlackHole 路由** | TTS 音频无法到达会议参会者 | 多输出设备间接路由 |

#!/bin/bash
# 会议实时语音对话 — 持续轮询 wrapper
# 每 3 秒调用 meeting_voice.py 拉字幕 + 自动回复

echo "   会议ID: ${MEETING_ID:-自动检测}"
echo "   按 Ctrl+C 退出"

while true; do
    if [ -n "$MEETING_ID" ]; then
        python3 "$SCRIPT_DIR/meeting_voice.py" "$MEETING_ID" 2>&1
    else
        python3 "$SCRIPT_DIR/meeting_voice.py" 2>&1
    fi
    sleep 3
done
